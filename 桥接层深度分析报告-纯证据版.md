## 1. Mutator 机制

### 1.1 核心代码

**文件**: `loot-core/src/server/mutators.ts`

```typescript
// 标记处理器为变异操作
const mutatingMethods = new WeakMap();

export function mutator<T extends HandlerFunctions>(handler: T): T {
  mutatingMethods.set(handler, true);
  return handler;
}

// 核心执行逻辑
export async function runHandler<T extends Handlers[keyof Handlers]>(
  handler: T,
  args?: Parameters<T>[0],
  { undoTag, name }: { undoTag?; name? } = {},
): Promise<ReturnType<T>> {
  if (mutatingMethods.has(handler)) {
    return runMutator(() => handler(args), { undoTag });
  }

  // 非 Mutator 方法直接执行
  const promise = handler(args);
  runningMethods.add(promise);
  void promise.then(() => {
    runningMethods.delete(promise);
  });
  return promise;
}

// runMutator 被 sequential 包装，确保串行执行
export const runMutator = sequential(_runMutator) as typeof _runMutator;

function _runMutator<T extends () => Promise<unknown>>(
  func: T,
  initialContext = {},
): Promise<Awaited<ReturnType<T>>> {
  currentContext = initialContext;
  return func().finally(() => {
    currentContext = null;
  }) as Promise<Awaited<ReturnType<T>>>;
}
```

### 1.2 sequential 队列机制

**文件**: `loot-core/src/shared/async.ts`

```typescript
export function sequential<T extends AnyFunction>(
  fn: T,
): (...args: Parameters<T>) => Promise<Awaited<ReturnType<T>>> {
  const sequenceState: {
    running: Promise<Awaited<ReturnType<T>>> | null;
    queue: Array<{ args; resolve; reject }>;
  } = { running: null, queue: [] };

  function pump() {
    const next = sequenceState.queue.shift();
    if (next !== undefined) {
      run(next.args, next.resolve, next.reject);
    } else {
      sequenceState.running = null;
    }
  }

  function run(args, resolve, reject) {
    sequenceState.running = fn.apply(null, args).then(
      val => { pump(); resolve(val); },
      err => { pump(); reject(err); },
    );
  }

  return ((...args) => {
    if (!sequenceState.running) {
      return new Promise((resolve, reject) => run(args, resolve, reject));
    } else {
      return new Promise((resolve, reject) => {
        sequenceState.queue.push({ resolve, reject, args });
      });
    }
  }) as T;
}
```

async function uploadFileWeb({ filename, contents }) {
  // 证据 1: 非浏览器直接返回
  if (!Platform.isBrowser) {
    return null;
  }

  // 证据 2: 文件名过滤
  const safeName = filename.split(/[/\\]/).pop()?.replaceAll('\0', '');
  if (!safeName || safeName === '.' || safeName === '..') {
    throw new Error('Invalid upload filename');
  }

  await fs.writeFile(fs.join('/uploads', safeName), contents);
  return {};
}

### 4.2 saveSyncedPrefs 完整时序代码

**文件**: `packages/desktop-client/src/prefs/prefsSlice.ts:120-132`

```typescript
export const saveSyncedPrefs = createAppAsyncThunk(
  `${sliceName}/saveSyncedPrefs`,
  async ({ prefs }: SaveSyncedPrefsPayload, { dispatch }) => {
    // 第 1 步：等待所有远程请求完成
    await Promise.all(
      Object.entries(prefs).map(([prefName, value]) =>
        send('preferences/save', {
          id: prefName as keyof SyncedPrefs,
          value,
        }),
      ),
    );
    // 第 2 步：所有 send 成功后，更新本地 Redux 状态
    dispatch(mergeSyncedPrefs(prefs));
  },
);
```

> **结论**: saveSyncedPrefs = 先 await 所有 send 请求成功，再更新本地状态

## 5. 最终结论矩阵

| 事项 | 纯代码证据结论 |
|------|---------------|
| Mutator 作用 | 顺序执行 + Undo 上下文设置/清理 |
| Mutator 不包含 | 数据库 BEGIN/COMMIT/ROLLBACK 事务逻辑 |
| saveSyncedPrefs 时序 | await Promise.all(send) 全部成功 → dispatch merge |
| upload-file-web 安全 | 1. 非浏览器直接 return null 2. 文件名 split + null 字节过滤 + . / .. 检查 |
| close-budget 特殊 | 唯一调用 flushRunningMethods 等待所有运行中方法 |
| 预算写操作模式 | 各方法独立 try-catch 策略，无统一模式 |
